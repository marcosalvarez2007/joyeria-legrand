-- ---------- PRODUCTOS ----------
create table if not exists productos (
  id bigint generated always as identity primary key,
  nombre text not null,
  descripcion_corta text,
  descripcion text,
  precio numeric(12,2) not null default 0 check (precio >= 0),
  categoria_id bigint references categorias(id) on delete set null,
  imagenes text[] default '{}',
  specs jsonb default '{}'::jsonb,
  stock integer default 0 check (stock >= 0),
  min_stock integer default 5,
  stock_note text,
  activo boolean default true,
  destacado boolean default false,
  nuevo boolean default false,
  tags text[] default '{}',
  creado_en timestamptz default now(),
  updated_at timestamptz default now()
);

create index if not exists idx_productos_categoria on productos(categoria_id);
create index if not exists idx_productos_activo on productos(activo);

-- FIX: updated_at ya no se queda quieto cuando se edita un producto
create or replace function set_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

drop trigger if exists trg_productos_updated_at on productos;
create trigger trg_productos_updated_at
  before update on productos
  for each row execute function set_updated_at();

-- ---------- USUARIOS (perfil, vinculado 1 a 1 con auth.users) ----------
create table if not exists usuarios (
  id uuid primary key references auth.users(id) on delete cascade,
  email text,
  full_name text,
  phone text,
  address text,
  gender text,
  birthday date,
  avatar_url text,
  role text not null default 'user' check (role in ('user','admin')),
  created_at timestamptz default now()
);

create or replace function handle_new_user()
returns trigger
language plpgsql
security definer set search_path = public
as $$
begin
  insert into public.usuarios (id, email, full_name, phone, address, gender, birthday)
  values (
    new.id,
    new.email,
    new.raw_user_meta_data->>'full_name',
    new.raw_user_meta_data->>'phone',
    new.raw_user_meta_data->>'address',
    new.raw_user_meta_data->>'gender',
    nullif(new.raw_user_meta_data->>'birthday','')::date
  )
  on conflict (id) do nothing;
  return new;
end;
$$;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function handle_new_user();

-- ---------- PEDIDO_ITEMS ----------
create table if not exists pedido_items (
  id bigint generated always as identity primary key,
  pedido_id bigint references pedidos(id) on delete cascade,
  producto_id bigint references productos(id) on delete set null,
  cantidad integer not null default 1 check (cantidad > 0),
  precio_unitario numeric(12,2) not null default 0
);

create index if not exists idx_pedido_items_pedido on pedido_items(pedido_id);
create index if not exists idx_pedido_items_producto on pedido_items(producto_id);

-- FIX: el precio ya no lo manda el cliente. Se toma siempre del
-- catálogo real al momento de crear el ítem, así nadie puede pagar
-- un precio inventado. Además descuenta stock automáticamente.
create or replace function set_precio_y_descontar_stock()
returns trigger
language plpgsql
security definer set search_path = public
as $$
declare
  precio_real numeric(12,2);
  stock_actual integer;
begin
  select precio, stock into precio_real, stock_actual
  from productos where id = new.producto_id for update;

  if precio_real is null then
    raise exception 'Producto % no existe o no está disponible', new.producto_id;
  end if;

  if stock_actual is not null and stock_actual < new.cantidad then
    raise exception 'Stock insuficiente para el producto %', new.producto_id;
  end if;

  new.precio_unitario = precio_real;

  update productos set stock = stock - new.cantidad where id = new.producto_id;

  return new;
end;
$$;

drop trigger if exists trg_pedido_items_precio_stock on pedido_items;
create trigger trg_pedido_items_precio_stock
  before insert on pedido_items
  for each row execute function set_precio_y_descontar_stock();

-- FIX: si se cancela/borra un ítem, el stock vuelve
create or replace function devolver_stock()
returns trigger
language plpgsql
security definer set search_path = public
as $$
begin
  update productos set stock = stock + old.cantidad where id = old.producto_id;
  return old;
end;
$$;

drop trigger if exists trg_pedido_items_devolver_stock on pedido_items;
create trigger trg_pedido_items_devolver_stock
  after delete on pedido_items
  for each row execute function devolver_stock();

create or replace function is_admin()
returns boolean
language sql
security definer set search_path = public
as $$
  select exists (
    select 1 from usuarios where id = auth.uid() and role = 'admin'
  );
$$;

  for select using (activo = true or is_admin());

  for all using (is_admin()) with check (is_admin());

  for update using (is_admin());

  for update using (auth.uid() = usuario_id and estado = 'pendiente')
  with check (estado = 'cancelado');
